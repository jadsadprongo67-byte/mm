        function hexToRgb(hex) {
            let result = /^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i.exec(hex);
            return result ? { r: parseInt(result[1], 16), g: parseInt(result[2], 16), b: parseInt(result[3], 16) } : null;
        }

        function getLuminance(r, g, b) {
            let a = [r, g, b].map(function (v) {
                v /= 255;
                return v <= 0.03928 ? v / 12.92 : Math.pow((v + 0.055) / 1.055, 2.4);
            });
            return a[0] * 0.2126 + a[1] * 0.7152 + a[2] * 0.0722;
        }

        function getContrast(l1, l2) {
            let lightest = Math.max(l1, l2);
            let darkest = Math.min(l1, l2);
            return (lightest + 0.05) / (darkest + 0.05);
        }

        function initS2() {
            s2BaseImg = generateBaseImage();
            const canvas = document.getElementById('s2-canvas');
            const ctx = canvas.getContext('2d');
            ctx.imageSmoothingEnabled = false; // Nearest neighbor upscaling

            const resIn = document.getElementById('s2-res');
            const bitIn = document.getElementById('s2-bit');

                state.s2.res = parseInt(resIn.value);

                document.getElementById('s2-res-val').textContent = state.s2.res + '%';
                document.getElementById('s2-bit-val').textContent = state.s2.bit + ' bit';

                let w = Math.max(1, Math.floor(1280 * (state.s2.res / 100)));
                let h = Math.max(1, Math.floor(720 * (state.s2.res / 100)));

                document.getElementById('s2-dim').textContent = `${w} x ${h} px`;
                
                let colors = Math.pow(2, state.s2.bit * 3);
                document.getElementById('s2-colors').textContent = formatNum(colors) + ' สี';
                
                let rawSize = Math.floor(w * h * state.s2.bit * 3 / 8);
                let mb = (rawSize / 1048576).toFixed(2);
                document.getElementById('s2-size').textContent = `${formatNum(rawSize)} ไบต์ (~${mb} MB)`;

                // Process image via JS manually for true quantization and nearest neighbor downscale
                const outData = new ImageData(w, h);
                const levels = Math.pow(2, state.s2.bit) - 1;
                const srcData = s2BaseImg.data;
                const destData = outData.data;

                let scaleX = 1280 / w;
                let scaleY = 720 / h;

                // Simple nearest neighbor & quantize
                for (let y = 0; y < h; y++) {
                    let srcY = Math.floor(y * scaleY);
                    for (let x = 0; x < w; x++) {
                        let srcX = Math.floor(x * scaleX);
                        let srcIdx = (srcY * 1280 + srcX) * 4;
                        let destIdx = (y * w + x) * 4;

                        // Quantize R, G, B
                        for(let c=0; c<3; c++) {
                            let val = srcData[srcIdx + c];
                            // map to 0-1, multiply by levels, round, map back to 0-255
                            destData[destIdx + c] = Math.round(Math.round((val / 255) * levels) * (255 / levels));
                        }
                        destData[destIdx + 3] = 255; // Alpha
                    }
                }

                // Draw to temp, then scale up to main
                const tempC = document.createElement('canvas');
                tempC.width = w; tempC.height = h;
                tempC.getContext('2d').putImageData(outData, 0, 0);

                ctx.clearRect(0, 0, 1280, 720);
                ctx.drawImage(tempC, 0, 0, w, h, 0, 0, 1280, 720);

                // Mission: Raw < 1048576, bit == 8, w >= 640
                let success = (rawSize < 1048576) && (state.s2.bit === 8) && (w >= 640);
                checkMission(2, success);
            };

            resIn.addEventListener('input', update);
            bitIn.addEventListener('input', update);
            update();
        }

        function initS3() {
            const rateIn = document.getElementById('s3-rate');
            const bitIn = document.getElementById('s3-bit');
            const playBtn = document.getElementById('s3-play');
            const msgEl = document.getElementById('s3-audio-msg');
            const canvas = document.getElementById('s3-canvas');
            const ctx = canvas.getContext('2d');
            const w = canvas.width;
            const h = canvas.height;
            const freq = 440;

            const update = () => {
                state.s3.rate = parseInt(rateIn.value);
                state.s3.bit = parseInt(bitIn.value);

                document.getElementById('s3-bit-val').textContent = state.s3.bit + ' bit';
                
                let nyquist = state.s3.rate > (2 * freq);
                const elNyq = document.getElementById('s3-nyquist');
                elNyq.className = nyquist ? 'pass-tag pass' : 'pass-tag fail';
                elNyq.textContent = nyquist ? 'ผ่าน (ไม่เกิด Aliasing)' : 'ไม่ผ่าน (เกิด Aliasing)';

                let levels = Math.pow(2, state.s3.bit);
                document.getElementById('s3-levels').textContent = formatNum(levels) + ' ระดับ';

                let rawSize = (state.s3.rate * state.s3.bit) / 8;
                let kb = (rawSize / 1024).toFixed(1);
                document.getElementById('s3-size').textContent = `${formatNum(rawSize)} B/s (~${kb} KB/s)`;

                // Draw Wave (2 periods)
                ctx.clearRect(0,0,w,h);
                let duration = 2 / freq; // ~0.0045s
                
                // Draw axes
                ctx.strokeStyle = "rgba(0,0,0,0.1)";
                ctx.beginPath(); ctx.moveTo(0, h/2); ctx.lineTo(w, h/2); ctx.stroke();

                // Draw Original smooth wave (thin)
                ctx.strokeStyle = "rgba(0,0,0,0.2)";
                ctx.lineWidth = 1;
                ctx.beginPath();
                for(let x=0; x<w; x++) {
                    let t = (x / w) * duration;
                    let val = Math.sin(2 * Math.PI * freq * t);
                    let y = h/2 - (val * h/2.5);
                    if(x===0) ctx.moveTo(x,y); else ctx.lineTo(x,y);
                }
                ctx.stroke();

                // Draw Quantized/Sampled wave
                ctx.strokeStyle = varColor('--primary', '#0064e4');
                ctx.fillStyle = varColor('--primary', '#0064e4');
                ctx.lineWidth = 2;
                
                let numSamples = Math.floor(state.s3.rate * duration);
                let stepX = w / numSamples;
                
                let maxLevel = levels - 1;
                
                ctx.beginPath();
                let lastY = h/2;
                let lastX = 0;

                for(let i=0; i<=numSamples; i++) {
                    let t = i / state.s3.rate;
                    if (t > duration) break;
                    let cx = (t / duration) * w;
                    
                    let realVal = Math.sin(2 * Math.PI * freq * t); // -1 to 1
                    let normVal = (realVal + 1) / 2; // 0 to 1
                    let quantVal = Math.round(normVal * maxLevel) / maxLevel;
                    let finalVal = (quantVal * 2) - 1;
                    
                    let cy = h/2 - (finalVal * h/2.5);

                    if(i===0) {
                        ctx.moveTo(cx, cy);
                    } else {
                        // Step interpolation for drawing
                        ctx.lineTo(cx, lastY);
                        ctx.lineTo(cx, cy);
                    }
                    lastX = cx; lastY = cy;

                    // Draw sample dot
                    ctx.fillRect(cx-2, cy-2, 4, 4);
                }
                ctx.stroke();

                // Mission: Nyquist=T, bit>=8, rawSize<=20480
                let success = nyquist && (state.s3.bit >= 8) && (rawSize <= 20480);
                checkMission(3, success);
                
                if(state.s3.rate < 8000) {
                    playBtn.disabled = true; msgEl.style.display = 'block';
                } else {
                    playBtn.disabled = false; msgEl.style.display = 'none';
                }
            };

            const varColor = (name, fallback) => getComputedStyle(document.documentElement).getPropertyValue(name).trim() || fallback;

            const playAudio = () => {
                if(state.s3.rate < 8000) return;
                
                if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
                if(audioCtx.state === 'suspended') audioCtx.resume();
                
                // Create buffer of 1 second
                // Note: browser AudioContext might not support all random rates (like 11025). 
                // We create at native rate, but generate the sine math as if it was sampled at lower rate
                
                let nativeRate = audioCtx.sampleRate;
                let buffer = audioCtx.createBuffer(1, nativeRate, nativeRate);
                let data = buffer.getChannelData(0);
                
                let maxLevel = Math.pow(2, state.s3.bit) - 1;
                
                for(let i=0; i<nativeRate; i++) {
                    let t = i / nativeRate;
                    // simulate low sample rate by discretizing time
                    let sampledT = Math.floor(t * state.s3.rate) / state.s3.rate; 
                    
                    let realVal = Math.sin(2 * Math.PI * freq * sampledT);
                    let normVal = (realVal + 1) / 2;
                    let quantVal = Math.round(normVal * maxLevel) / maxLevel;
                    let finalVal = (quantVal * 2) - 1;
                    
                    data[i] = finalVal * 0.2; // 0.2 volume to not be too loud
                }

                let source = audioCtx.createBufferSource();
                source.buffer = buffer;
                source.connect(audioCtx.destination);
                source.start();
                
                playBtn.textContent = "กำลังเล่น...";
                playBtn.disabled = true;
                setTimeout(() => {
                    playBtn.textContent = "▶ เล่นเสียง (1 วินาที)";
                    playBtn.disabled = false;
                }, 1000);
            };

            rateIn.addEventListener('change', update);
            bitIn.addEventListener('input', update);
            playBtn.addEventListener('click', playAudio);
            update();
        }

        let s4LastTime = 0;
        let s4Ball = { x: 300, y: 150, vx: 200, vy: 150, r: 20 };
        function s4Loop(time) {
            // Only run if active tab
            if (!document.getElementById('station-4').classList.contains('active')) return;
            
            const canvas = document.getElementById('s4-canvas');
            const ctx = canvas.getContext('2d');
            let w = canvas.width; let h = canvas.height;

            let dt = (time - s4LastTime) / 1000;
            let targetDt = 1 / state.s4.fps;

            // Only update and draw if enough time passed for the target fps
            if (dt >= targetDt || reducedMotion) {
                if(!reducedMotion) {
                    s4Ball.x += s4Ball.vx * dt;
                    s4Ball.y += s4Ball.vy * dt;
                    
                    if(s4Ball.x - s4Ball.r < 0) { s4Ball.x = s4Ball.r; s4Ball.vx *= -1; }
                    if(s4Ball.x + s4Ball.r > w) { s4Ball.x = w - s4Ball.r; s4Ball.vx *= -1; }
                    if(s4Ball.y - s4Ball.r < 0) { s4Ball.y = s4Ball.r; s4Ball.vy *= -1; }
                    if(s4Ball.y + s4Ball.r > h) { s4Ball.y = h - s4Ball.r; s4Ball.vy *= -1; }
                }

                ctx.fillStyle = '#222';
                ctx.fillRect(0, 0, w, h);
                
                // Draw Ball
                ctx.beginPath();
                ctx.arc(s4Ball.x, s4Ball.y, s4Ball.r, 0, Math.PI*2);
                ctx.fillStyle = '#ff4757';
                ctx.fill();
                
                // Draw Frame Info
                ctx.fillStyle = 'rgba(255,255,255,0.7)';
                ctx.font = '14px monospace';
                ctx.fillText(`FPS: ${state.s4.fps}`, 10, 20);
                if(reducedMotion) ctx.fillText(`[Reduced Motion]`, 10, 40);

                s4LastTime = time;
            }
            if(!reducedMotion) requestAnimationFrame(s4Loop);
        }

        function initS5() {
            const container = document.getElementById('quiz-container');
            const resetBtn = document.getElementById('quiz-reset');
            
            const renderQuiz = () => {
                container.innerHTML = '';
                let allCorrect = true;

                questions.forEach((q, i) => {
                    const qDiv = document.createElement('div');
                    qDiv.className = 'quiz-q';
                    qDiv.innerHTML = `<p>${q.q}</p>`;
                    
                    const status = state.s5[`q${i+1}`];
                    if (status !== 1) allCorrect = false;

                    q.choices.forEach((c, j) => {
                        const label = document.createElement('label');
                        label.className = 'quiz-choice';
                        // Disable if already answered correctly
                        if(status === 1) label.style.pointerEvents = 'none';

                        const radio = document.createElement('input');
                        radio.type = 'radio';
                        radio.name = `q${i}`;
                        radio.value = j;
                        radio.disabled = status === 1;
                        
                        radio.addEventListener('change', () => {
                            if (j === q.ans) {
                                state.s5[`q${i+1}`] = 1;
                            } else {
                                state.s5[`q${i+1}`] = -1;
                            }
                            renderQuiz();
                        });

                        label.appendChild(radio);
                        label.appendChild(document.createTextNode(c));
                        qDiv.appendChild(label);
                    });

                    // Feedback box
                    const fb = document.createElement('div');
                    fb.className = 'quiz-feedback';
                    if (status === 1) {
                        fb.classList.add('show', 'correct');
                        fb.innerHTML = `<strong>ถูกต้อง!</strong> ${q.exp}`;
                    } else if (status === -1) {
                        fb.classList.add('show', 'wrong');
                        fb.innerHTML = `<strong>ยังไม่ถูก ลองคิดดูใหม่นะ</strong>`;
                    }
                    qDiv.appendChild(fb);
                    container.appendChild(qDiv);
                });

                if (allCorrect) {
                    checkMission(5, true);
                    resetBtn.style.display = 'block';
                }
            };

            resetBtn.addEventListener('click', () => {
                state.s5 = { q1: 0, q2: 0, q3: 0, q4: 0, q5: 0 };
                resetBtn.style.display = 'none';
                renderQuiz();
            });

            renderQuiz();
        }

        window.addEventListener('DOMContentLoaded', () => {
            loadProgress();
            initS1();
            initS2();
            initS3();
            initS4();
            initS5();
        });

    </script>
